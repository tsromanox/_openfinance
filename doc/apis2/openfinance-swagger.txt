  securitySchemes:
    OAuth2:

        clientCredentials:
          tokenUrl: https://auth.openfinance.com.br/token
          scopes:
            consents:read: Leitura de consentimentos
    ApiKey:
      type: apiKey
      in: header
      name: X-API-Key

  parameters:
    consentId:
      name: consentId
      in: path
      description: ID do consentimento
      required: true
      schema:
        type: string
        pattern: '^[a-zA-Z0-9][a-zA-Z0-9-]{0,99}$'
        maxLength: 100

    Authorization:
      name: Authorization
      in: header
      description: Token OAuth2 Bearer
      required: true
      schema:
        type: string
        pattern: '^Bearer .+$'

    xFapiAuthDate:
      name: x-fapi-auth-date
      in: header
      description: Data/hora da autenticação do usuário
      required: false
      schema:
        type: string

    xFapiCustomerIpAddress:
      name: x-fapi-customer-ip-address
      in: header
      description: Endereço IP do cliente
      required: false
      schema:
        type: string
        pattern: '^(?:[0-9]{1,3}\.){3}[0-9]{1,3}$'

    xFapiInteractionId:
      name: x-fapi-interaction-id
      in: header
      description: UUID único para rastreabilidade
      required: true
      schema:
        type: string
        format: uuid

    xCustomerUserAgent:
      name: x-customer-user-agent
      in: header
      description: User Agent do cliente
      required: false
      schema:
        type: string

    page:
      name: page
      in: query
      description: Número da página (paginação)
      schema:
        type: integer

        minimum: 1

    pageSize:

      in: query
      description: Quantidade de registros por página

        type: integer
        default: 25
        minimum: 1
        maximum: 100

        status:
          type: string
          description: Estado atual do consentimento
          enum:
            - AWAITING_AUTHORISATION
            - AUTHORISED
            - REJECTED
            - CONSUMED
        statusUpdateDateTime:
          type: string
          format: date-time
          description: Data e hora da última atualização do status
        permissions:
          type: array
          description: Permissões concedidas
          items:
            type: string
            enum:
              - ACCOUNTS_READ
              - ACCOUNTS_BALANCES_READ
              - ACCOUNTS_TRANSACTIONS_READ
              - ACCOUNTS_OVERDRAFT_LIMITS_READ
              - CREDIT_CARDS_ACCOUNTS_READ
              - CREDIT_CARDS_ACCOUNTS_BILLS_READ
              - CREDIT_CARDS_ACCOUNTS_BILLS_TRANSACTIONS_READ
              - CREDIT_CARDS_ACCOUNTS_LIMITS_READ
              - CREDIT_CARDS_ACCOUNTS_TRANSACTIONS_READ
        expirationDateTime:
          type: string
          format: date-time
          description: Data de expiração do consentimento
        transactionFromDateTime:
          type: string
          format: date-time
          description: Data inicial para consulta de transações
        transactionToDateTime:
          type: string
          format: date-time
          description: Data final para consulta de transações
        rejectionReason:
          $ref: '#/components/schemas/RejectionReason'
        businessEntity:
          $ref: '#/components/schemas/BusinessEntity'
        loggedUser:
          $ref: '#/components/schemas/LoggedUser'

      type: object
      required:

        - type
        - status
      properties:

          type: string
          description: ID do recurso
        type:
          type: string
          description: Tipo do recurso
          enum:
            - ACCOUNT
            - CREDIT_CARD_ACCOUNT
            - LOAN
            - FINANCING
            - INVOICE_FINANCING
            - OVERDRAFT
            - INVESTMENT
        status:
          type: string
          description: Status do recurso
          enum:
            - AVAILABLE
            - UNAVAILABLE
            - PENDING_AUTHORISATION
        statusUpdateDateTime:
          type: string
          format: date-time

      type: object
      properties:
        code:
          type: string
          enum:
            - CUSTOMER_MANUALLY_REJECTED
            - CONSENT_EXPIRED
            - CONSENT_MAXIMAL_EXPIRATION_EXCEEDED
            - CONSENT_REVOKED
        detail:
          type: string
          maxLength: 2048

    BusinessEntity:
      type: object
      properties:
        document:
          type: object
          properties:
            identification:
              type: string
              pattern: '^\d{14}
            rel:
              type: string

    LoggedUser:
      type: object
      properties:
        document:
          type: object
          properties:
            identification:
              type: string
              pattern: '^\d{11}
            rel:
              type: string

          type: string

        prev:
          type: string
          format: uri
        next:
          type: string
          format: uri
        last:
          type: string
          format: uri

      required:
        - totalRecords

        totalRecords:
          type: integer
          description: Total de registros
        totalPages:
          type: integer
          description: Total de páginas
        requestDateTime:
          type: string
          format: date-time

      type: object
      required:

        requestDateTime:
          type: string
          format: date-time

    Error:
      type: object
      required:
        - errors
      properties:
        errors:

            properties:
              code:
                type: string
              title:
                type: string
              detail:
                type: string
        meta:
          $ref: '#/components/schemas/MetaSingle'
